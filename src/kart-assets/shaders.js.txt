/* Native GLES2/WebGL1 shaders. Procedural surfaces keep the build self-contained. */
(() => {
  const vertex=`
attribute vec3 aPosition;attribute vec3 aNormal;attribute vec3 aColour;attribute float aMaterial;
uniform mat4 uViewProjection,uModel,uLightVP;uniform vec3 uEye;uniform float uSkyPass;
varying vec3 vColour,vNormal,vWorld;varying float vMaterial,vDistance;varying vec4 vShadow;
void main(){if(uSkyPass>.5){vWorld=aPosition;gl_Position=vec4(aPosition.xy,1.,1.);return;}vec4 world=uModel*vec4(aPosition,1.0);vWorld=world.xyz;vNormal=normalize(mat3(uModel)*aNormal);vColour=aColour;vMaterial=aMaterial;vDistance=distance(world.xyz,uEye);vShadow=uLightVP*world;gl_Position=uViewProjection*world;}`;
  const fragment=`
precision highp float;
varying vec3 vColour,vNormal,vWorld;varying float vMaterial,vDistance;varying vec4 vShadow;
uniform vec3 uFog,uEye;uniform float uAlpha,uTime,uShadowPass,uShadows,uSkyPass;uniform sampler2D uShadowMap;
float hash(vec2 p){return fract(sin(dot(p,vec2(127.1,311.7)))*43758.5453);}
float noise(vec2 p){vec2 i=floor(p),f=fract(p);f=f*f*(3.0-2.0*f);return mix(mix(hash(i),hash(i+vec2(1.,0.)),f.x),mix(hash(i+vec2(0.,1.)),hash(i+vec2(1.)),f.x),f.y);}
vec4 packDepth(float d){vec4 v=fract(d*vec4(16777216.,65536.,256.,1.));return v-v.xxyz*vec4(0.,1./256.,1./256.,1./256.);}
float depthAt(vec2 uv){return dot(texture2D(uShadowMap,uv),vec4(1./16777216.,1./65536.,1./256.,1.));}
float shadow(vec3 n){if(uShadows<.5)return 1.;vec3 p=vShadow.xyz/vShadow.w*.5+.5;if(p.x<0.||p.x>1.||p.y<0.||p.y>1.||p.z>1.)return 1.;float bias=max(.00024,.00080*(1.-dot(n,normalize(vec3(-.4,.85,.3)))));float light=0.;for(int x=-1;x<=1;x++)for(int y=-1;y<=1;y++){float d=depthAt(p.xy+vec2(float(x),float(y))/1024.);light+=p.z-bias<=d?1.:.49;}return light/9.;}
void main(){
  if(uSkyPass>.5){vec2 uv=vWorld.xy*.5+.5;vec3 sky=mix(uFog*1.10,uFog*vec3(.87,.94,1.02),uv.y);vec2 p=uv*vec2(4.2,5.5)+vec2(uTime*.002,0.);float cloud=noise(p)*.56+noise(p*2.1)*.28+noise(p*4.3)*.16;float mask=smoothstep(.54,.78,cloud)*smoothstep(.36,.85,uv.y);gl_FragColor=vec4(mix(sky,vec3(1.,.99,.96),mask*.86),1.);return;}
  float distanceToEye=distance(vWorld,uEye);
  if(uShadowPass>.5){gl_FragColor=packDepth(gl_FragCoord.z);return;}
  vec3 n=normalize(vNormal),base=vColour,view=normalize(uEye-vWorld),sun=normalize(vec3(-.4,.85,.3));float rough=.85,shine=.05;
  if(vMaterial>.5&&vMaterial<1.5){float grain=hash(floor(vWorld.xz*26.));float patches=noise(vWorld.xz*.19);base*=.96+.045*grain+.045*patches;rough=.90;shine=.08;}
  else if(vMaterial>1.5&&vMaterial<2.5){float horizontal=dot(vWorld.xz,vec2(-n.z,n.x));vec2 uv=vec2(horizontal,vWorld.y)*vec2(1.65,4.2);float row=floor(uv.y);uv.x+=mod(row,2.)*.5;vec2 brick=fract(uv);float mortar=max(step(brick.x,.048),step(brick.y,.095));base=mix(base*(.86+.22*hash(floor(uv))),vec3(.47,.46,.40),mortar*.48);}
  else if(vMaterial>2.5&&vMaterial<3.5){base*=.87+.20*noise(vec2(vWorld.y*4.,(vWorld.x+vWorld.z)*18.));}
  else if(vMaterial>3.5&&vMaterial<4.5){base*=.90+.12*noise(vWorld.xz*.31)+.035*hash(floor(vWorld.xz*9.));}
  else if(vMaterial>4.5&&vMaterial<5.5){float fresnel=pow(1.-max(dot(n,view),0.),3.);vec3 sky=mix(vec3(.39,.49,.56),uFog,.68);base=mix(base,sky,.25+fresnel*.60);base*=.94+.06*sin(vWorld.y*2.);rough=.19;shine=.48;}
  else if(vMaterial>5.5&&vMaterial<6.5){n=normalize(n+vec3(sin(vWorld.x*.72+uTime)*.10,0.,cos(vWorld.z*.83+uTime*.8)*.12));float fresnel=pow(1.-max(dot(n,view),0.),3.);base=mix(base,uFog,.24+fresnel*.48);base*=.93+.07*sin(vWorld.x*1.7+vWorld.z*1.4+uTime*1.8);rough=.18;shine=.65;}
  else if(vMaterial>6.5&&vMaterial<7.5){rough=.26;shine=.30;}
  else if(vMaterial>7.5&&vMaterial<8.5){gl_FragColor=vec4(mix(base,uFog,smoothstep(140.,620.,distanceToEye)),uAlpha);return;}
  else if(vMaterial>9.5){float leaf=noise(vWorld.xz*5.+vWorld.yy*1.8);base*=.91+.16*leaf;n=normalize(n+vec3((leaf-.5)*.17,.02,(leaf-.5)*.11));}
  else if(vMaterial>8.5){float grain=hash(floor(vWorld.xz*34.+vWorld.yy*7.));base*=.95+grain*.08;}
  float diffuse=max(dot(n,sun),0.),occlusion=shadow(n);vec3 hemi=mix(vec3(.36,.36,.30),vec3(.58,.65,.72),n.y*.5+.5);
  vec3 colour=base*(hemi+vec3(.66,.62,.49)*diffuse*occlusion);
  float spec=pow(max(dot(n,normalize(sun+view)),0.),mix(110.,5.,rough));colour+=vec3(1.,.93,.78)*spec*shine*occlusion;
  colour=mix(colour,uFog,smoothstep(130.,600.,distanceToEye));colour=pow(max(colour,vec3(0.)),vec3(.93));
  gl_FragColor=vec4(clamp(colour,0.,1.),uAlpha);
}`;
  globalThis.KartShaders={vertex,fragment};
})();
