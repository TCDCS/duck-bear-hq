/* Optional screen-relative orientation input. No sensor data is stored or sent. */
(() => {
  'use strict';
  class WackyTilt {
    constructor(){this.enabled=false;this.ready=false;this.status='Tilt is off. Buttons still work.';this.sensitivity=1;this.last=0;this.value=0;this.neutral=0;this.raw=0;this.angle=null;this.env=null;this.listener=null;}
    sample(e,angle=0,now=Date.now()){
      if(!this.enabled||!Number.isFinite(e.beta)||!Number.isFinite(e.gamma))return;
      angle=((angle%360)+360)%360;
      const radians=angle*Math.PI/180,raw=e.gamma*Math.cos(radians)+e.beta*Math.sin(radians);
      this.raw=raw;
      if(!this.ready||this.angle!==angle){this.neutral=raw;this.value=0;this.ready=true;this.angle=angle;}
      const delta=((raw-this.neutral+540)%360)-180,dead=2.5;
      const target=Math.max(-1,Math.min(1,Math.sign(delta)*Math.max(0,Math.abs(delta)-dead)/24*this.sensitivity));
      const dt=Math.min(.1,Math.max(.001,(now-this.last)/1000));this.value+=(target-this.value)*(1-Math.exp(-dt*14));
      this.last=now;this.status='Tilt steering on. Hold comfortably, then calibrate.';
    }
    read(now=Date.now()){return this.enabled&&this.ready&&now-this.last<650?this.value:0;}
    calibrate(){this.neutral=this.raw;this.value=0;this.status=this.ready?'Centred. Tilt left or right to steer.':'Waiting for a sensor reading. Touch controls still work.';}
    async enable(env=globalThis){
      if(this.enabled)return true;
      if(!env.isSecureContext){this.status='Tilt needs a secure HTTPS page. Use the buttons here.';return false;}
      if(!env.DeviceOrientationEvent){this.status='This browser does not expose orientation sensors. Use the buttons.';return false;}
      try{
        if(typeof env.DeviceOrientationEvent.requestPermission==='function'&&await env.DeviceOrientationEvent.requestPermission()!=='granted'){this.status='Motion permission denied. Touch controls still work.';return false;}
        this.enabled=true;this.ready=false;this.value=0;this.env=env;
        this.listener=e=>this.sample(e,env.screen?.orientation?.angle??env.orientation??0,Date.now());
        env.addEventListener('deviceorientation',this.listener,{passive:true});
        this.status='Waiting for the sensor. Hold your phone comfortably.';return true;
      }catch{this.enabled=false;this.status='Motion permission could not be enabled. Use the buttons.';return false;}
    }
    disable(){if(this.listener)this.env?.removeEventListener('deviceorientation',this.listener);this.enabled=false;this.ready=false;this.value=0;this.listener=null;this.status='Tilt is off. Buttons still work.';}
  }
  globalThis.WackyTilt=WackyTilt;
})();
